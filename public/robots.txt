# CRM закрыта от сканирования целиком, открыта только публичная онлайн-запись
# (/book). Правило Allow длиннее Disallow, поэтому для /book выигрывает оно.
#
# Ассеты открыты намеренно: без JS и CSS краулер не отрендерит витрину, а без
# доступа к картинке не соберётся превью ссылки в WhatsApp/Facebook — их
# краулер (facebookexternalhit) уважает robots.txt.
#
# /media/organizations/ — логотипы клиник: бэкенд отдаёт краулерам собственные
# og-теги и ставит в og:image ссылку на logo_url оттуда. Без этого правила
# картинка запрещена и карточка ссылки приходит без логотипа. Открыт ровно
# этот подкаталог: в остальном /media/ лежат документы и фото пациентов.
#
# /media/staff/employees/ — фото врачей: на странице врача (/book/doctor/…)
# бэкенд ставит в og:image его фото, и без правила превью приходит без картинки.
# Эти фото и так публичны — их показывает сама витрина.
#
# /static/public_api/ — логотип платформы для нейтральной карточки CRM: на
# любой адрес CRM, кроме /book и /site, бэкенд отдаёт краулерам og-теги
# платформы с этим og:image. Из статики Django открыт только этот подкаталог.
User-agent: *
Disallow: /
Allow: /book
Allow: /assets/
Allow: /media/organizations/
Allow: /media/staff/employees/
Allow: /static/public_api/
Allow: /og-image.png
Allow: /booking-icon.png
Allow: /favicon.ico
Allow: /favicon.svg
